# training/requirements-training.txt
# ====================================
# Training environment dependencies for AMPVNet + AdaFace pipeline.
#
# IMPORTANT: These are NOT the Pi 5 deployment requirements.
#   Pi 5 runtime uses: app/requirements.txt (fastapi, opencv, mediapipe, onnxruntime)
#   This file is ONLY for the machine you train on (desktop/cloud GPU or CPU).
#
# Install with:
#   pip install -r training/requirements-training.txt
#
# Recommended training environment:
#   Python 3.10+, CUDA 12.x (if GPU available), 16GB+ RAM
#   Training on Pi 5 CPU is possible but slow (~30 min/epoch for CASIA).

torch>=2.1.0               # PyTorch: AMPVNet backbone, AdaFace loss, training loop

# ---- Computer Vision ----

# ---- Scientific Computing ----

# ---- Model Evaluation ----

                           # (optional but strongly recommended)
